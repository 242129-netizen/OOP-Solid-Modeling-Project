void calculation(string name, string rollno, float q1, float q2, float q3,
                 float q4, float a1, float a2, float m, float f, float p)
{
    cout << endl;

    // Use float instead of int
    float total =
        (((q1 + q2 + q3 + q4) / 40.0) * 10.0)
        + (((a1 + a2) / 10.0) * 5.0)
        + ((m / 50.0) * 30.0)
        + ((f / 100.0) * 45.0)
        + ((p / 40.0) * 10.0);

    cout << setw(26) << name
         << setw(10) << rollno
         << setw(10) << q1
         << setw(10) << q2
         << setw(10) << q3
         << setw(10) << q4
         << setw(10) << a1
         << setw(10) << a2
         << setw(10) << m
         << setw(10) << f
         << setw(10) << p
         << setw(15) << total
         << endl;

    cout << "__________________________________________________________________________________________________________________________________";
}

// Convert String To Float
float convertStringToInt(string str)
{
    float num;

    stringstream ss(str);
    ss >> num;

    return num;
}
